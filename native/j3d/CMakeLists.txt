cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(gc_wii_j3d_assets LANGUAGES CXX)
  enable_testing()
endif()
add_library(gc_wii_j3d_assets STATIC asset.cpp)
target_compile_features(gc_wii_j3d_assets PUBLIC cxx_std_20)
target_include_directories(gc_wii_j3d_assets PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(test_j3d_assets test_asset.cpp)
target_link_libraries(test_j3d_assets PRIVATE gc_wii_j3d_assets)
add_test(NAME j3d_resource_import_rom_free COMMAND test_j3d_assets)
set(GC_WII_J3D_GXCORE_INCLUDE_DIR "" CACHE PATH "Optional host shader backend headers for native J3D materials")
if(GC_WII_J3D_GXCORE_INCLUDE_DIR)
  add_library(gc_wii_j3d_materials STATIC material.cpp)
  target_link_libraries(gc_wii_j3d_materials PUBLIC gc_wii_j3d_assets)
  target_include_directories(gc_wii_j3d_materials PUBLIC "${GC_WII_J3D_GXCORE_INCLUDE_DIR}")
  add_executable(test_j3d_materials test_material.cpp)
  target_link_libraries(test_j3d_materials PRIVATE gc_wii_j3d_materials)
  add_test(NAME j3d_material_compiler_rom_free COMMAND test_j3d_materials)
endif()
